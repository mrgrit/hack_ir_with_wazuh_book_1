# C10-02 | kt66 VM · ccc 호스트 셸
# ls -ld는 디렉터리 자체의 존재·권한을 본다. is-enabled는 부팅 시 자동 시작 설정을 확인하며 active와 별개이다. iptables -S는 패킷 처리 규칙을 읽는다. INPUT은 호스트로 들어오는 패킷의 체인, KT66-PERIMETER는 kt66의 사용자 체인이다.

# C10-03 | kt66 VM · ccc 호스트 셸
# mkdir는 디렉터리를 만들며 -m 700은 소유자만 접근하도록 한다. cp -p는 파일과 권한·시각을 보존해 복사한다. sha256sum은 내용의 SHA-256 해시를 출력한다. 줄 끝 역슬래시(\)는 다음 줄을 같은 명령으로 잇는다.

sudo cp -p /var/ossec/etc/ossec.conf \
~/book10-state/ossec.conf.before
sudo sha256sum /var/ossec/etc/ossec.conf

# C10-04 | kt66 VM · ccc 호스트 셸
# cat은 파일 내용을 표준 출력으로 읽는다. 첫 파일은 적용 중인 설정, 두 번째는 책의 기준 커밋에 포함된 설정 원본이다.

cat ~/kt66/deployment/osquery-linux.conf

# C10-05 | kt66 VM · ccc 호스트 셸
# install은 지정한 권한으로 파일을 복사한다. 0644는 소유자 읽기·쓰기, 나머지 읽기 권한이다. --config_path는 검사할 설정, --config_check는 설정 검사이다.
sudo install -m 0644 \
~/kt66/deployment/osquery-linux.conf \
/etc/osquery/osquery.conf
sudo osqueryi \
--config_path=/etc/osquery/osquery.conf --config_check
# 확인: 검사가 오류 없이 끝나야 한다. 기존 설정 파일이 있는 환경에서는 먼저 별도로 백업하고 필요한 schedule 항목만 병합한다.

# C10-06 | kt66 VM · ccc 호스트 셸
# sed는 텍스트에서 지정 범위를 출력한다. -n은 자동 출력을 끄고, 쉼표 앞뒤 검색식이 시작·끝 줄이며 p는 출력이다. 작은따옴표는 셸 해석을 막고 \/는 검색식 안의 슬래시 문자이다. grep -n은 줄 번호, -F는 정규표현식 없이 문자열 그대로 찾는 옵션이다.
sudo sed -n '/<wodle name="osquery">/,/<\/wodle>/p' \
/var/ossec/etc/ossec.conf
sudo grep -n -F '/var/log/kern.log' \
/var/ossec/etc/ossec.conf
# 확인: osquery 블록은 disabled=no, run_daemon=no여야 한다. kern.log를 읽는 localfile도 확인한다.

# C10-07 | kt66 VM · ccc 호스트 셸
# nano는 터미널 파일 편집기이다. 기존 osquery 블록을 아래 내용과 대조하여 수정한다. kern.log localfile이 없으면 ossec_config의 닫는 태그 앞에 추가한다. Ctrl+O와 Enter로 저장하고 Ctrl+X로 닫는다.
sudo nano /var/ossec/etc/ossec.conf
# 확인: 같은 osquery 블록이나 kern.log 수집 항목을 중복으로 추가하지 않는다. 주소·등록 정보 등 나머지 설정은 유지한다.

# C10-08 | kt66 VM · ccc 호스트 셸
# 두 Wazuh 실행 파일의 -t는 설정 검사이다. 모두 오류 없이 끝난 뒤 start로 osqueryd를 시작하고 restart로 에이전트를 재시작한다.

sudo /var/ossec/bin/wazuh-modulesd -t

# C10-09 | kt66 VM · ccc 호스트 셸

# C10-10 | kt66 VM · ccc 호스트 셸
# ss는 소켓 정보를 보여 준다. -l은 리스닝, -n은 주소·포트를 숫자로 표시, -t는 TCP, -p는 프로세스 정보이다. 작은따옴표의 sport = :31337은 로컬 포트 조건이다.
sudo ss -lntp 'sport = :31337'
# 확인: 표 머리글만 나오면 해당 포트의 TCP 리스너가 없다. 기존 프로세스가 나오면 종료하지 말고 다른 빈 실습 포트를 정한다.

# C10-11 | kt66 VM · ccc 호스트 셸 · 터미널 A
# -l은 연결 대기, -k는 연결이 끝나도 계속 대기하는 옵션이다. IP는 연결을 받을 로컬 주소, 31337은 포트이다. 터미널 A에서 실행한 채 둔다.
nc -lk 192.168.0.107 31337
# 확인: 프롬프트가 돌아오지 않는 상태가 정상이다. Address already in use는 포트 충돌이다. 새 명령은 터미널 B에서 입력한다.

# C10-13 | kt66 VM · ccc 호스트 셸

# C10-14 | kt66 VM · ccc 호스트 셸
# iptables -S로 INPUT에서 사용자 체인으로 이동하는 규칙과 체인 내부 순서를 읽는다. ip -4 route get은 상대 주소로 나갈 IPv4 경로를 표시한다. dev 뒤는 네트워크 인터페이스이다. cat으로 실행할 kt66 스크립트 내용을 먼저 확인한다.
sudo iptables -S INPUT
sudo iptables -S KT66-PERIMETER
ip -4 route get 192.168.0.108
cat ~/kt66/deployment/perimeter.sh

# C10-15 | kt66 VM · ccc 호스트 셸
# iptables-save는 현재 규칙을 텍스트로 출력한다. 꺾쇠(>)는 그 출력을 파일에 저장하며 같은 이름이 있으면 덮어쓴다. bash는 확인한 스크립트를 실행한다.

# C10-19 | 공격자 VM · att 셸
# -sT는 TCP 연결 스캔, -Pn은 별도 호스트 발견 검사를 생략하고 대상이 응답한다고 가정하는 옵션이다. -n은 이름 조회를 생략한다. -p 1-1000은 포트 범위, --max-retries 1은 탐색 재전송의 상한이다.
date -Is
nmap -sT -Pn -n -p 1-1000 \
--max-retries 1 192.168.0.107
# 확인: Nmap 종료 때까지 기다린다. 이 실습에서 허가된 대상 한 대의 주소인지 확인한다. -Pn의 Host is up 표시를 독립적인 생존 확인 결과로 읽지 않는다.

# C10-20 | kt66 VM · ccc 호스트 셸

sudo grep -F 'SRC=192.168.0.108 ' \
/var/log/kern.log | tail -n 5
# 확인: 스캔 시각의 DROP, SRC, DST, SPT, DPT, PROTO와 SYN을 기록한다. 공격자 터미널이 UTC이면 그 시각에 9시간을 더해 한국 시각과 맞춘다.

docker exec -it kt66-siem /bin/bash
# 확인: 프롬프트가 컨테이너로 바뀐 뒤 다음 파일을 읽는다.

exit

# C10-23 | kt66 VM · ccc 호스트 셸
# INPUT에서 이번에 추가한 정확한 규칙을 먼저 확인한다. -D는 그 규칙 삭제, -i는 들어오는 인터페이스, -j는 이동 대상이다. -F는 지정 체인의 규칙 비우기, -X는 비어 있고 참조되지 않는 사용자 체인 삭제이다. ens33은 자기 환경 값으로 바꾼다.
sudo iptables -S INPUT
sudo iptables -D INPUT -i ens33 -j KT66-PERIMETER
sudo iptables -F KT66-PERIMETER
sudo iptables -X KT66-PERIMETER
sudo iptables -S INPUT
# 확인: INPUT이 사전 기록과 같은지 확인한다. 다른 체인 이름을 넣거나 체인 이름을 생략해 전체 규칙을 비우지 않는다.

# C10-25 | kt66 VM · ccc 호스트 셸

# 확인: 둘 중 하나가 있으면 내용을 확인하고 이전 실습을 정리하거나 새 파일명을 정한다. 기존 파일 위에 전송·백업하지 않는다.

# C10-26 | 공격자 VM · att 셸
# ssh-keygen은 키 쌍을 생성하고 지문을 확인하는 도구이다. 두 파일이 없을 때만 생성한다. -t는 알고리즘, -N 뒤 빈 작은따옴표는 실습 키의 빈 암호문, -C는 주석, -f는 저장 경로이다. -l과 -f는 지정한 공개 키의 지문 출력이다.
ls -l ~/book10-key ~/book10-key.pub
ssh-keygen -t ed25519 -N '' \
-C book10-key@attacker -f ~/book10-key
ssh-keygen -lf ~/book10-key.pub
# 확인: 기존 파일이 있으면 덮어쓰지 않는다. .pub가 공개 키이며 지문과 주석을 기록한다.

# C10-27 | 공격자 VM · att 셸
# scp는 SSH 연결로 파일을 복사한다. 콜론 앞은 대상 계정·주소, 뒤는 원격 홈 기준 저장 이름이다. 서버 호스트 키 지문을 kt66 콘솔에서 확인한 값과 대조한 뒤 ccc의 호스트 로그인 암호를 입력한다.
scp ~/book10-key.pub \
ccc@192.168.0.107:book10-key.pub
# 확인: kt66 호스트의 /home/ccc/book10-key.pub에 공개 키만 전송됐는지 확인한다. 개인 키 파일은 전송하지 않는다.

# C10-28 | kt66 VM · ccc 호스트 셸
# ls로 대상과 소유자를 확인한다. 전송한 키의 지문을 공격자 VM의 출력과 비교하고 authorized_keys의 초기 해시를 기록한다.
ls -ld ~/.ssh
ls -l ~/.ssh/authorized_keys ~/book10-key.pub
ssh-keygen -lf ~/book10-key.pub
sha256sum ~/.ssh/authorized_keys
# 확인: 기존 authorized_keys가 있으면 다음 백업을 수행한다. 파일이 없으면 초기 상태를 “없음”으로 기록하고 아래 준비를 먼저 수행한다.

# C10-29 | kt66 VM · ccc 호스트 셸
# authorized_keys가 없는 경우에만 수행한다. mkdir -p는 없는 디렉터리를 만들고 touch는 없는 파일을 빈 파일로 만든다. chmod는 권한을 설정한다. 700은 소유자만 디렉터리 접근, 600은 소유자만 파일 읽기·쓰기를 허용한다. 기존 .ssh의 권한은 먼저 기록한다.
mkdir -p ~/.ssh
chmod 700 ~/.ssh
touch ~/.ssh/authorized_keys
chmod 600 ~/.ssh/authorized_keys
# 확인: ls -ld ~/.ssh와 ls -l ~/.ssh/authorized_keys로 ccc 소유·권한을 확인한다. 빈 파일을 만들기 전 초기 상태가 없음이었다는 기록을 남긴다.

# C10-30 | kt66 VM · ccc 호스트 셸
# 백업 이름이 없는 것을 확인한 뒤 cp -p로 원본을 보존한다. cmp는 두 파일의 바이트를 비교한다.
