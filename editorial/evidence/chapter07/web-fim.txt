<syscheck>
    <disabled>no</disabled>
    <frequency>300</frequency>
    <scan_on_start>yes</scan_on_start>
    <!-- 채점 관심사 = '변경'(modified). 초기 스캔의 기존 파일마다 알림(alert_new_files)을
         켜면 CRS 등 다수 파일에서 첫 기동 시 알림 폭주 → no 로 둔다(기본값). -->
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/apache2</directories>
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/modsecurity</directories>
    <directories realtime="yes" report_changes="yes">/usr/share/modsecurity-crs</directories>
    <directories realtime="yes" report_changes="yes">/home/ccc</directories>
    <!-- 지속성이 실제로 심기는 자리. /etc 전체는 12시간 주기 스캔이라 한 교시 안에
         절대 안 뜬다. 흔적은 남겨 놓고 SIEM 이 조용하면 "SIEM 으로 찾는다"는 실습이
         성립하지 않는다. 그래서 이 경로들만 실시간으로 본다. -->
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/cron.d</directories>
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/sudoers.d</directories>
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/ld.so.preload</directories>
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/passwd</directories>
    <!-- 강의 보강 2: 계정·권한 파일, 사용자 crontab, 시스템 바이너리는 실시간. -->
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/shadow</directories>
    <directories realtime="yes" report_changes="yes" whodata="yes">/etc/sudoers</directories>
    <directories realtime="yes" report_changes="yes" whodata="yes">/var/spool/cron</directories>
    <directories realtime="yes">/usr/bin,/usr/sbin</directories>
    <!-- /var/log 는 계속 커지므로 내용 변경은 보지 않고 파일 생성·삭제만 본다(로그 삭제 탐지). -->
    <directories realtime="yes" check_all="no">/var/log</directories>
